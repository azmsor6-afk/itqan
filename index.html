/* دالة تجريد النص العثماني ومعالجة الحروف المقطعة لضمان دقة التطابق */ function normalizeArabic(text) { if (!text) return ''; let cleaned = text .replace(/[\u0610-\u061A\u064B-\u065F\u0670\u06D6-\u06DC\u06DF-\u06E8\u06EA-\u06ED]/g, '') .replace(/[أإآٱ]/g, 'ا') .replace(/ة/g, 'ه') .replace(/ى/g, 'ي') .replace(/[^\w\sء-ي]/gi, '') .trim(); cleaned = cleaned .replace(/\bالف لام ميم\b/g, 'الم') .replace(/\bالف لام را\b/g, 'الر') .replace(/\bالف لام ميم ص\b/g, 'المص') .replace(/\bالف لام ميم را\b/g, 'المر') .replace(/\bكاف ها يا عين صاد\b/g, 'كهيعص') .replace(/\bطا سين ميم\b/g, 'طسم') .replace(/\bطا سين\b/g, 'طس') .replace(/\bحاميم\b/g, 'حم') .replace(/\bعين سين قاف\b/g, 'عسق'); return cleaned; } /* تهيئة التعرف الصوتي باستخدام Web Speech API */ if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) { const SR = window.SpeechRecognition || window.webkitSpeechRecognition; recognition = new SR(); recognition.continuous = true; recognition.interimResults = true; recognition.lang = 'ar-SA'; recognition.onresult = (e) => { let text = ''; for (let i = e.resultIndex; i < e.results.length; i++) text += e.results[i][0].transcript; document.getElementById('transcriptTextSmall').innerText = text; processSpokenText(text); }; recognition.onend = () => { if (isListening) recognition.start(); }; } /* معالجة النص المنطوق ومطابقته مع الكلمات المخفية */ function processSpokenText(transcript) { if (!isTestMode) return; const cleanSpoken = normalizeArabic(transcript); const spokenWords = cleanSpoken.split(/\s+/).filter(w => w.length > 0); const blurredSpans = Array.from(document.querySelectorAll('.word.blurred')); if (blurredSpans.length === 0) return; spokenWords.forEach(spoken => { if (spoken.length < 1) return; for (let i = 0; i < Math.min(5, blurredSpans.length); i++) { const target = blurredSpans[i]; if (!target.classList.contains('blurred')) continue; const cleanTarget = target.dataset.clean; if (cleanTarget && (cleanTarget === spoken || cleanTarget.includes(spoken) || spoken.includes(cleanTarget))) { target.classList.remove('blurred'); target.classList.add('revealed-mic'); revealedWords++; break; } } }); updateAccuracy(); } /* تشغيل أو إيقاف المايك */ function toggleMic() { if (!recognition) return alert('الرجاء استخدام متصفح يدعم التعرف الصوتي.'); const btn = document.getElementById('micBtnSmall'); if (isListening) { recognition.stop(); isListening = false; btn.classList.remove('listening'); } else { recognition.start(); isListening = true; btn.classList.add('listening'); } }
